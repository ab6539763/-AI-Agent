一、上午：为什么从裸字典升级到面向对象（OOP）？

在大型智能体平台中，一条对话消息不仅有静态数据（role、content、name），还有内聚的业务行为（角色合法性校验、API 字典序列化、Token 消耗估算）。面向对象编程（OOP）的核心价值在于数据与行为的封装以及Fail-Fast（快速失败）：

  • 封装（Encapsulation）：将字段校验逻辑收敛在 __init__ 中，非法角色在构造瞬间直接抛出 ValueError，绝不流入下游大模型接口。
  • 三种方法分工明确：to_dict(self) 与 estimate_tokens(self) 操作实例状态；from_dict(cls, data) 负责从外部字典工厂构造实例；is_valid_role(role) 作为无状态静态校验器。

from typing import Any


class ChatMessage:
    """大模型标准对话消息实体类。"""

    VALID_ROLES: tuple[str, ...] = ("system", "user", "assistant", "tool")

    def __init__(self, role: str, content: str, name: str | None = None) -> None:
        if not self.is_valid_role(role):
            allowed = ", ".join(self.VALID_ROLES)
            raise ValueError(f"无效的消息角色: {role!r}，合法角色仅限: {allowed}")

        if not isinstance(content, str):
            raise TypeError(f"消息内容 content 必须为 str，实际收到: {type(content).__name__}")

        if name is not None and not isinstance(name, str):
            raise TypeError(f"可选字段 name 必须为 str 或 None，实际收到: {type(name).__name__}")

        self.role: str = role
        self.content: str = content
        self.name: str | None = name

    def to_dict(self) -> dict[str, Any]:
        """实例方法：将 ChatMessage 序列化为 OpenAI / DeepSeek API 兼容字典。"""
        payload: dict[str, Any] = {"role": self.role, "content": self.content}
        if self.name is not None:
            payload["name"] = self.name
        return payload

    def estimate_tokens(self) -> int:
        """实例方法：启发式估算当前消息文本的 Token 消耗量。"""
        if not self.content:
            return 0
        cjk_count = sum(1 for ch in self.content if "\u4e00" <= ch <= "\u9fff")
        other_count = len(self.content) - cjk_count
        return max(1, round(cjk_count * 1.5 + other_count * 0.3))

    @classmethod
    def from_dict(cls, data: dict[str, Any]) -> "ChatMessage":
        """类方法（工厂方法）：从字典反序列化构造 ChatMessage 实例。"""
        if not isinstance(data, dict):
            raise ValueError("from_dict 参数必须为 dict 类型")
        if "role" not in data or "content" not in data:
            raise ValueError("消息字典必须包含 'role' 和 'content' 字段")
        return cls(role=data["role"], content=data["content"], name=data.get("name"))

    @staticmethod
    def is_valid_role(role: str) -> bool:
        """静态方法：无需实例化即可校验角色字符串是否合法。"""
        return isinstance(role, str) and role in ChatMessage.VALID_ROLES

    def __repr__(self) -> str:
        name_part = f", name={self.name!r}" if self.name is not None else ""
        return f"ChatMessage(role={self.role!r}, content={self.content!r}{name_part})"

    def __eq__(self, other: object) -> bool:
        if not isinstance(other, ChatMessage):
            return NotImplemented
        return self.role == other.role and self.content == other.content and self.name == other.name


if __name__ == "__main__":
    print("静态校验 'user':", ChatMessage.is_valid_role("user"))
    print("静态校验 'admin':", ChatMessage.is_valid_role("admin"))

    msg = ChatMessage(role="user", content="请总结智枢平台 Day 08 架构升级。", name="chenmo")
    print("实例表示:", repr(msg))

    assert restored == msg

四、课堂笔记：OOP 三类方法与魔术方法速查表

语法特性装饰器 / 第一个参数在 ChatMessage 中的职责
构造函数 __init__self接收 role / content / name 并执行 Fail-Fast 合法性校验
实例方法无装饰器 / selfto_dict() 导出字典、estimate_tokens() 估算文本 Token
类方法@classmethod / clsfrom_dict(cls, data) 从 JSON 字典工厂构造对象实例
静态方法@staticmethod / 无隐式参数is_valid_role(role) 纯工具校验，无需实例化即可调用
魔术方法__repr__ / __eq__提供可读日志输出与基于字段值的对象相等性比较

# 作业 1 参考答案：Conversation 容器类
class Conversation:
    def __init__(self, system_prompt: str | None = None) -> None:
        self.messages: list[ChatMessage] = []
        if system_prompt is not None:
            self.messages.append(ChatMessage(role="system", content=system_prompt))

    def add_message(self, message: ChatMessage) -> None:
        if not isinstance(message, ChatMessage):
            raise TypeError(f"必须传入 ChatMessage 实例，实际收到: {type(message).__name__}")
        self.messages.append(message)

    def total_tokens(self) -> int:
        return sum(msg.estimate_tokens() for msg in self.messages)

    def to_dict_list(self) -> list[dict]:
        return [msg.to_dict() for msg in self.messages]


# 作业 2 参考答案：@classmethod 快捷工厂方法
class ExtendedChatMessage(ChatMessage):
    @classmethod
    def system(cls, content: str) -> "ExtendedChatMessage":
        return cls(role="system", content=content)

    @classmethod
    def user(cls, content: str, name: str | None = None) -> "ExtendedChatMessage":
        return cls(role="user", content=content, name=name)

    @classmethod
    def assistant(cls, content: str) -> "ExtendedChatMessage":
        return cls(role="assistant", content=content)


# 作业 3 参考答案：@staticmethod 合规校验工具类
class MessageValidator:
    @staticmethod
    def is_non_empty_content(content: str) -> bool:
        return isinstance(content, str) and len(content.strip()) > 0

    @staticmethod
    def is_within_length(content: str, max_length: int = 4000) -> bool:
        return isinstance(content, str) and len(content) <= max_length

    @staticmethod
    def validate_payload(data: dict, max_length: int = 4000) -> bool:
        if not isinstance(data, dict):
            return False
        role = data.get("role")
        content = data.get("content")
        return (
            ChatMessage.is_valid_role(role)
            and MessageValidator.is_non_empty_content(content)
            and MessageValidator.is_within_length(content, max_length=max_length)
        )
