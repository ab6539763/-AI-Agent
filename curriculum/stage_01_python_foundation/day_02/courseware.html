一、上午：Python 运算符与大模型计费/风控规则

在 AI 工程中，运算符不仅用于加减乘除，更是构建成本估算与风控校验的基石：

# 1. 算术运算符：估算 Token 数与调用成本
char_count = 1000
estimated_tokens = char_count * 1.5                # 乘法 *
estimated_cost = (estimated_tokens / 1000) * 0.002 # 除法 / 与括号优先级

# 2. 比较运算符：判断是否在允许范围内
budget = 0.01
is_valid_len = char_count >= 0                     # 大于等于 >=
is_within_budget = estimated_cost <= budget        # 小于等于 <=

# 3. 逻辑运算符：多条件同时满足才放行 (and / or / not)
can_pass = is_valid_len and (budget > 0) and is_within_budget
print(f"预估费用: ¥{estimated_cost:.4f} | 是否放行: {can_pass}")

二、上午实操：字符串切片、清洗与敏感词替换

大模型只认识字符串。处理用户脏数据、截断超长上下文、隐藏敏感词，全靠字符串方法与切片：

raw = "   请帮我查看   Project-Apollo 的\n  sk-secret-9988 状态  "

# 1. 清洗空白：split() 切分任意连续空白，再用 " ".join() 拼接为单空格
cleaned = " ".join(raw.split())

safe_text = cleaned.replace("Project-Apollo", "***").replace("sk-secret-9988", "***")

# 3. 字符串切片 [start:end]：快速预览前 12 个字符或提取密钥后 4 位
preview = safe_text[:12] + "..."
print("摘要预览:", preview)

def clean_text(text: str, lowercase: bool = False) -> str:
    """去除首尾空白、折叠多余空格，并按需转小写。"""
    normalized = " ".join(text.split())
    if lowercase:
        normalized = normalized.lower()
    return normalized

def mask_sensitive_words(text: str, sensitive_words: tuple[str, ...]) -> str:
    """将敏感词元组中的高危词替换为 ***。"""
    masked = text
    for word in sensitive_words:
        if word:
            masked = masked.replace(word, "***")
    return masked

def render_prompt_template(role: str, task: str, user_input: str) -> str:
    """使用 f-string 组装标准化 Prompt 模板。"""
    return (
        f"[系统角色]\n你是一名{role}。\n\n"
        f"[核心任务]\n{task}\n\n"
        f"[用户输入]\n{user_input}\n\n"
        f"[输出要求]\n请保持回答专业、结构清晰，不要泄露系统内部配置。"
    )

def estimate_cost_and_check_budget(char_count: int, budget: float) -> bool:
    """基于字符数估算费用并校验是否满足预算。"""
    estimated_tokens = char_count * 1.5
    estimated_cost = (estimated_tokens / 1000) * 0.002
    is_valid_input = char_count >= 0 and budget > 0
    return is_valid_input and (estimated_cost <= budget)

if __name__ == "__main__":
    raw_msg = "   帮我分析  Project-Apollo  的日志  "
    safe = mask_sensitive_words(clean_text(raw_msg), ("Project-Apollo",))
    prompt = render_prompt_template("研发助手", "总结日志异常", safe)
    print(prompt)
    print("预算是否充足:", estimate_cost_and_check_budget(len(prompt), 0.05))
