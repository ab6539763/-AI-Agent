from collections.abc import Callable
from typing import Any


def build_api_payload(
    model: str,
    *messages: dict[str, Any],
    temperature: float = 0.7,
    **extra_params: Any,
) -> dict[str, Any]:
    """动态构造 OpenAI / DeepSeek Chat Completions API 请求体字典。"""
    if not isinstance(model, str) or not model.strip():
        raise ValueError("model 必须是非空字符串")
    if not (0.0 <= temperature <= 2.0):
        raise ValueError(f"temperature 必须在 [0.0, 2.0] 范围内，当前为: {temperature}")
    for idx, msg in enumerate(messages):
        if not isinstance(msg, dict):
            raise TypeError(f"messages 第 {idx} 项必须是 dict 类型")

    payload: dict[str, Any] = {
        "model": model.strip(),
        "messages": list(messages),
        "temperature": temperature,
    }
    payload.update(extra_params)
    return payload


def apply_pipeline(text: str, *transforms: Callable[[str], str]) -> str:
    """高阶函数：按顺序将任意多个文本变换函数串联应用到输入文本上。"""
    if not isinstance(text, str):
        raise TypeError("text 必须是 str 类型")
    result = text
    for idx, fn in enumerate(transforms):
        if not callable(fn):
            raise TypeError(f"transforms 第 {idx} 项不可调用: {fn!r}")
        result = fn(result)
    return result


def count_nested_tokens(node: Any) -> int:
    """递归遍历嵌套 JSON / 消息树，统计所有文本节点的字符数（估算 Token）。"""
    if isinstance(node, str):
        return len(node)
    if isinstance(node, list):
        return sum(count_nested_tokens(item) for item in node)
    if isinstance(node, dict):
        return sum(count_nested_tokens(value) for value in node.values())
    return 0


if __name__ == "__main__":
    cleaned = apply_pipeline(
        "  请总结  【机密】Q3财报  ",
        str.strip,
        lambda s: " ".join(s.split()),
        lambda s: s.replace("【机密】", "[已脱敏]"),
    )
    payload = build_api_payload(
        "deepseek-chat",
        {"role": "user", "content": cleaned},
        temperature=0.3,
        max_tokens=2048,
        stream=True,
    )
    print("清洗结果:", cleaned)
    print("嵌套 Token 估算:", count_nested_tokens(payload))

def find_values_by_key(node, target_key: str) -> list:
    results = []
    if isinstance(node, dict):
        for key, value in node.items():
            if key == target_key:
                results.append(value)
            results.extend(find_values_by_key(value, target_key))
    elif isinstance(node, list):
        for item in node:
            results.extend(find_values_by_key(item, target_key))
    return results
