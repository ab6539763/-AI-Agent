二、核心知识点精讲：条件嵌套与循环控制

1. 嵌套 if / elif / else：权限分级防线

外层先判断“是否已开通大模型权限（llm_enabled）”，内层再根据工龄与预算划分 VIP、STANDARD 与 DENIED。先守住总闸门，再细分等级，能有效防止逻辑漏洞。

2. for 循环中的 continue 与 break

  • continue：跳过当前这一轮（例如遇到空字符串 Prompt），直接进入下一轮循环。
  • break：立即终止整个循环（例如在批处理队列中遇到 "STOP" 停止词，或用户在 while True 菜单输入 "0"）。

def check_access_level(years_exp: int, llm_enabled: bool, budget: float) -> str:
    """嵌套 if/elif/else 判断智枢平台访问权限等级。"""
    if llm_enabled:
        if years_exp >= 3 and budget >= 50.0:
            return "VIP"
        elif budget > 0 and (years_exp >= 1 or budget >= 10.0):
            return "STANDARD"
        else:
            return "DENIED"
    else:
        return "DENIED"


def batch_clean_prompts(prompts: list[str], stop_keyword: str = "STOP") -> list[str]:
    """for 循环批量清洗 Prompt：遇空跳过(continue)，遇停止词终止(break)。"""
    cleaned_list: list[str] = []
    for raw_prompt in prompts:
        stripped = raw_prompt.strip()
        if not stripped:
            continue                # 跳过空白 Prompt
        if stripped == stop_keyword:
            break                   # 遇到停止词立即退出批处理
        cleaned_list.append(" ".join(stripped.split()))
    return cleaned_list


def generate_multiplication_table(n: int = 9) -> str:
    """经典嵌套 for 循环生成下三角乘法口诀表。"""
    rows: list[str] = []
    for i in range(1, n + 1):
        cols = [f"{j}×{i}={i * j:<2}" for j in range(1, i + 1)]
        rows.append("  ".join(cols).rstrip())
    return "\n".join(rows)

# 作业 1：猜数字游戏逻辑
def guess_number_game(secret: int, guesses: list[int], max_attempts: int = 5) -> str:
    attempt = 0
    while attempt < max_attempts and attempt < len(guesses):
        guess = guesses[attempt]
        attempt += 1
        if guess == secret:
            return f"恭喜猜中！共用了 {attempt} 次。"
        elif guess > secret:
            print(f"第 {attempt} 次 ({guess}): 太大了")
        else:
            print(f"第 {attempt} 次 ({guess}): 太小了")
    return f"挑战失败，正确答案是 {secret}。"

# 作业 3A：两数之和 (Two Sum)
def two_sum(nums: list[int], target: int) -> list[int]:
    seen: dict[int, int] = {}
    for idx, num in enumerate(nums):
        if target - num in seen:
            return [seen[target - num], idx]
        seen[num] = idx
    return []


# 作业 3B：回文数 (Palindrome Number)
def is_palindrome(x: int) -> bool:
    if x < 0 or (x % 10 == 0 and x != 0):
        return False
    rev = 0
    while x > rev:
        rev = rev * 10 + x % 10
        x //= 10
    return x == rev or x == rev // 10
