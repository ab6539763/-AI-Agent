二、上午实操：装饰器 @retry_on_error 与 @log_latency

对应仓库 project_iterations/day_13/zhishu_agent/decorators.py，使用 @functools.wraps(func) 保留原函数元信息，结合 typing.TypeVar 保持类型提示完整：

import functools
import time
from typing import Any, Callable, TypeVar

F = TypeVar("F", bound=Callable[..., Any])

def retry_on_error(max_retries: int = 3, delay: float = 0.01) -> Callable[[F], F]:
    """异常自动重试装饰器。"""
    def decorator(func: F) -> F:
        @functools.wraps(func)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            last_exc: BaseException | None = None
            for attempt in range(1, max_retries + 1):
                try:
                    return func(*args, **kwargs)
                except Exception as exc:
                    last_exc = exc
                    if attempt < max_retries and delay > 0:
                        time.sleep(delay)
            assert last_exc is not None
            raise last_exc
        return wrapper  # type: ignore[return-value]
    return decorator

def log_latency(func: F) -> F:
    """函数执行耗时日志装饰器（单位：ms）。"""
    @functools.wraps(func)
    def wrapper(*args: Any, **kwargs: Any) -> Any:
        start = time.perf_counter()
        try:
            return func(*args, **kwargs)
        finally:
            elapsed_ms = (time.perf_counter() - start) * 1000
            print(f"[Latency] {func.__name__} 耗时 {elapsed_ms:.2f} ms")
    return wrapper  # type: ignore[return-value]

三、下午实操：生成器 yield 与 asyncio.gather 并发

对应仓库 project_iterations/day_13/zhishu_agent/streaming.py：

import asyncio
from collections.abc import Iterator

def stream_tokens(text: str, chunk_size: int = 1) -> Iterator[str]:
    """使用 yield 生成器按切片流式吐出文本 Token（惰性求值）。"""
    for idx in range(0, len(text), chunk_size):
        yield text[idx : idx + chunk_size]

async def _async_single_ask(prompt: str, model_name: str = "deepseek-chat", delay: float = 0.01) -> str:
    await asyncio.sleep(delay)
    return f"[{model_name}] 已异步处理: {prompt.strip()}"

async def async_batch_ask(prompts: list[str], model_name: str = "deepseek-chat", delay: float = 0.01) -> list[str]:
    """使用 asyncio.gather 并发执行多条 Prompt 请求，保持返回顺序。"""
    tasks = [_async_single_ask(p, model_name=model_name, delay=delay) for p in prompts]
    return list(await asyncio.gather(*tasks))

四、下午实操：零依赖 .env 安全配置与密钥脱敏

对应仓库 project_iterations/day_13/zhishu_agent/config.py，纯标准库解析 .env，且默认 override=False 不覆盖宿主机/CI 环境变量，并在 AgentConfig.__repr__ 中自动脱敏 api_key：

def mask_secret(secret: str, visible_tail: int = 4) -> str:
    if not secret:
        return "<EMPTY>"
    if len(secret) <= visible_tail:
        return "*" * len(secret)
    return f"{'*' * 4}...{secret[-visible_tail:]}"
