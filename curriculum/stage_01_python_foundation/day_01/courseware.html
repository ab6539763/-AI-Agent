四、下午实操：核心代码逐行解析（可直接粘贴运行）

以下代码对应仓库中的 project_iterations/day_01/zhishu_agent/profile.py：

import platform
import sys

def get_environment_info() -> dict[str, str]:
    """获取当前 Python 开发环境信息，用于新员工入职环境自检。"""
    return {
        "python_version": platform.python_version(),
        "os_info": f"{platform.system()} {platform.release()}",
        "executable": sys.executable,
    }

def build_employee_card(
    name: str,
    role: str,
    years_exp: int,
    hourly_cost: float,
    llm_enabled: bool,
) -> str:
    """根据基本数据类型生成格式化的企业数字名片。"""
    # 三元表达式：将布尔值转为中文友好的状态
    status_label = "已开通 [OK]" if llm_enabled else "未开通 [LOCKED]"
    border = "=" * 46
    lines = [
        border,
        "   星辰数科 · 智枢平台数字名片 (v0.1.0-day01)",
        border,
        f"姓名 / 标识 : {name}",
        f"所属岗位    : {role}",
        f"工程经验    : {years_exp} 年",
        f"资源预算    : ¥{hourly_cost:.2f} / 小时",
        f"大模型权限  : {status_label}",
        border,
    ]
    return "\n".join(lines)

if __name__ == "__main__":
    env = get_environment_info()
    print(f"[环境自检] Python {env['python_version']} | 系统: {env['os_info']}")
    print(build_employee_card("陈默", "AI应用开发工程师", 2, 12.5, True))
