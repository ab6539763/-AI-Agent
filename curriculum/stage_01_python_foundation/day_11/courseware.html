三、下午实操：核心代码逐行解析

以下代码对应仓库中的 project_iterations/day_11/zhishu_agent/doc_scanner.py 核心片段：

from pathlib import Path
from datetime import datetime, timezone
import csv, json, re

def scan_directory(root_dir: Path, keywords: list[str]) -> dict:
    # 1. 预编译正则，re.escape 防止关键词中的特殊符号（如 C++）报错
    patterns = {
        kw: re.compile(re.escape(kw), re.IGNORECASE)
        for kw in keywords
    }
    records = []
    # 2. pathlib 递归扫描并按相对路径排序，保证跨平台结果确定性
    for path in sorted(root_dir.rglob("*")):
        if not path.is_file() or path.suffix.lower() not in {".txt", ".md"}:
            continue
        text = path.read_text(encoding="utf-8", errors="replace")
        stat = path.stat()
        counts = {kw: len(pat.findall(text)) for kw, pat in patterns.items()}
        records.append({
            "relative_path": path.relative_to(root_dir).as_posix(),
            "size_bytes": stat.st_size,
            "modified_at": datetime.fromtimestamp(stat.st_mtime, tz=timezone.utc).isoformat(),
            "total_hits": sum(counts.values()),
            "keyword_counts": counts,
        })
    return {
        "scanned_at": datetime.now(timezone.utc).isoformat(),
        "total_files": len(records),
        "files": records,
    }

四、避坑指南：工程实践中的 4 个高频陷阱

  • 陷阱 1：CSV 写入出现空行 —— 在 Windows 下用 open(..., "w") 写 CSV 必须加 newline=""，否则 csv.DictWriter 会每行多写一个空行。
  • 陷阱 2：中文 JSON 变成 \uXXXX —— json.dumps 默认 ensure_ascii=True，务必显式传入 ensure_ascii=False 保留可读中文。
  • 陷阱 3：英文全词匹配误伤中文 —— 仅当关键词首尾为 \w（字母/数字/下划线）时才加 \b 边界；对符号或特殊词先 re.escape()。
  • 陷阱 4：忘记创建父目录 —— 导出报告前先调用 out_path.parent.mkdir(parents=True, exist_ok=True)，避免 FileNotFoundError。

import re

PHONE_RE = re.compile(r"(?<!\d)(1[3-9]\d)\d{4}(\d{4})(?!\d)")
EMAIL_RE = re.compile(r"\b([A-Za-z0-9])[A-Za-z0-9._%+-]*(@[A-Za-z0-9.-]+\.[A-Za-z]{2,})\b")

def mask_sensitive_info(text: str) -> str:
    text = PHONE_RE.sub(r"\1****\2", text)
    return EMAIL_RE.sub(r"\1***\2", text)
