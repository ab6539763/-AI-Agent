一、上午：三大容器的工程选型与对比

在企业级后端与 Agent 开发中，什么时候用 list，什么时候用 tuple 或 set，不是凭感觉，而是由可变性、有序性与唯一性决定的：

容器类型语法示例核心特性在智枢平台中的真实用途
列表 list[task1, task2]有序、可变、允许重复Agent 待办任务队列、对话历史消息列表、Top-K 切片
元组 tuple("deepseek-chat", "qwen-max")有序、不可变系统受支持大模型白名单、只读配置项、多返回值解包
集合 set{"RAG", "Python"}无序、自动去重多路 RAG 标签合并去重（|）、共同命中分析（&）

# 2. 集合去重与交集：处理多路 RAG 检索标签
def merge_unique_tags(tags_a: list[str], tags_b: list[str]) -> set[str]:
    set_a = {tag.strip() for tag in tags_a if tag.strip()}
    set_b = {tag.strip() for tag in tags_b if tag.strip()}
    return set_a | set_b

# 3. 列表任务队列：列表推导式筛选 + 切片 Top-K
class TodoManager:
    def __init__(self) -> None:
        self._tasks: list[dict] = []

    def add_task(self, title: str, priority: str = "normal") -> dict:
        task = {"title": title.strip(), "priority": priority, "done": False}
        self._tasks.append(task)
        return task

    def complete_task(self, title: str) -> bool:
        for task in self._tasks:
            if task["title"] == title.strip():
                task["done"] = True
                return True
        return False

    def get_pending_tasks(self) -> list[dict]:
        # 列表推导式：过滤所有未完成任务
        return [t for t in self._tasks if not t["done"]]

    def top_k_tasks(self, k: int) -> list[dict]:
        # 列表切片：提取前 k 个任务
        return self._tasks[:k] if k > 0 else []

if __name__ == "__main__":
