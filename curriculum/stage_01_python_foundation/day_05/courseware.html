import json
from pathlib import Path
from typing import Any


def extract_llm_response(raw_json: str) -> dict[str, Any]:
    """解析模拟的 OpenAI / DeepSeek API 响应 JSON 字符串。"""
    try:
        data = json.loads(raw_json)
    except json.JSONDecodeError as exc:
        raise ValueError(f"LLM 响应 JSON 解析失败: {exc}") from exc

    if not isinstance(data, dict):
        raise ValueError("LLM 响应根节点必须为 JSON 对象 (dict)")

    choices = data.get("choices")
    if not isinstance(choices, list) or len(choices) == 0:
        raise ValueError("LLM 响应缺少有效的 choices 列表")

    content = choices[0].get("message", {}).get("content", "")
    usage = data.get("usage", {})

    return {
        "id": str(data.get("id", "")),
        "model": str(data.get("model", "")),
        "content": str(content),
        "prompt_tokens": int(usage.get("prompt_tokens", 0)),
        "completion_tokens": int(usage.get("completion_tokens", 0)),
        "total_tokens": int(usage.get("total_tokens", 0)),
    }


def save_session_json(filepath: Path, messages: list[dict[str, Any]]) -> None:
    """将多轮对话保存为 JSON 文件（保留中文与缩进）。"""
    target = Path(filepath)
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open("w", encoding="utf-8") as f:
        json.dump(messages, f, ensure_ascii=False, indent=2)


def load_session_json(filepath: Path) -> list[dict[str, Any]]:
    """从本地 JSON 文件加载多轮对话历史。"""
    target = Path(filepath)
    if not target.exists():
        raise FileNotFoundError(f"会话文件不存在: {target}")
    with target.open("r", encoding="utf-8") as f:
        return json.load(f)


def summarize_usage(responses: list[dict[str, Any]]) -> dict[str, int]:
    """遍历多次响应字典，聚合统计 Token 消耗。"""
    total_prompt = total_completion = total_all = 0
    for resp in responses:
        usage = resp.get("usage", {})
        prompt = int(usage.get("prompt_tokens", 0))
        completion = int(usage.get("completion_tokens", 0))
        total = int(usage.get("total_tokens", prompt + completion))
        total_prompt += prompt
        total_completion += completion
        total_all += total
    return {
        "request_count": len(responses),
        "prompt_tokens": total_prompt,
        "completion_tokens": total_completion,
        "total_tokens": total_all,
    }

import json
from pathlib import Path

# 作业 1 参考答案：Function Calling 嵌套提取
def extract_tool_call(raw_json: str) -> dict:
    data = json.loads(raw_json)
    fn = data["choices"][0]["message"]["tool_calls"][0]["function"]
    return {
        "tool_name": fn["name"],
        "arguments": json.loads(fn["arguments"]),
    }

# 作业 2 参考答案：中文 JSON 序列化
config = {"agent_name": "智枢客服专员", "model": "deepseek-chat", "tags": ["企业知识库", "中文优先"]}
p = Path("agent_config.json")
with p.open("w", encoding="utf-8") as f:
    json.dump(config, f, ensure_ascii=False, indent=2)
with p.open("r", encoding="utf-8") as f:
    loaded = json.load(f)
assert loaded["agent_name"] == "智枢客服专员"

# 作业 3 参考答案：按模型聚合 Token 用量
def aggregate_usage_by_model(responses: list[dict]) -> dict[str, dict[str, int]]:
    stats: dict[str, dict[str, int]] = {}
    for resp in responses:
        model = resp.get("model", "unknown")
        tokens = int(resp.get("usage", {}).get("total_tokens", 0))
        if model not in stats:
            stats[model] = {"calls": 0, "total_tokens": 0}
        stats[model]["calls"] += 1
        stats[model]["total_tokens"] += tokens
    return stats
