四、下午实操：核心代码逐行解析（MockTransport 离线可运行）

以下代码对应仓库中的 project_iterations/day_12/zhishu_agent/llm_client.py 核心调用与测试模式：

import json
import httpx
from zhishu_agent.exceptions import AuthenticationError, RateLimitError, ServerError
from zhishu_agent.llm_client import LLMClient
from zhishu_agent.message import ChatMessage

# 1. 定义离线 MockTransport，100% 模拟云端大模型网关
def mock_gateway(request: httpx.Request) -> httpx.Response:
    if request.headers.get("Authorization") == "Bearer sk-invalid":
        return httpx.Response(401, json={"error": {"message": "Invalid API Key"}})
    payload = json.loads(request.content.decode("utf-8"))
    return httpx.Response(
        200,
        json={
            "choices": [{
                "message": {
                    "role": "assistant",
                    "content": f"[Mock:{payload['model']}] 收到：{payload['messages'][-1]['content']}"
                }
            }]
        },
    )

# 2. 注入 MockTransport 进行零成本离线调用
transport = httpx.MockTransport(mock_gateway)
with LLMClient(api_key="sk-demo", transport=transport) as client:
    reply = client.chat([ChatMessage(role="user", content="你好，智枢！")])
    print(reply)

五、课堂笔记：大模型常见 HTTP 状态码速查表

HTTP 状态码含义映射异常类企业级处理建议
200 OK请求成功并返回结果无解析 choices[0].message.content
400 Bad Request请求体参数或模型名错误LLMAPIError检查 JSON 字段、temperature 或上下文长度
401 UnauthorizedAPI Key 无效、过期或缺失AuthenticationError禁止重试，立即告警并检查环境变量
429 Too Many Requests触发限流（RPM/TPM）或余额不足RateLimitError读取 Retry-After，执行指数退避重试
500 / 502 / 503大模型服务端或网关临时故障ServerError记录日志，可有限次重试或切换备用厂商

def classify_http_status(status_code: int) -> dict[str, object]:
    if status_code == 200:
        return {"category": "请求成功", "should_retry": False}
    if status_code == 400:
        return {"category": "客户端请求参数错误", "should_retry": False}
    if status_code == 401:
        return {"category": "鉴权失败（API Key 无效或过期）", "should_retry": False}
    if status_code == 429:
        return {"category": "触发限流（RPM/TPM 超限）", "should_retry": True}
    if status_code in (500, 502, 503):
        return {"category": "服务端临时故障", "should_retry": True}
    return {"category": f"未知状态码 ({status_code})", "should_retry": False}

# 作业 2 参考答案
def build_deepseek_payload(system_prompt: str, user_prompt: str, model: str = "deepseek-chat", temperature: float = 0.7, max_tokens: int = 2048, stream: bool = False) -> dict[str, object]:
    if not system_prompt.strip() or not user_prompt.strip():
        raise ValueError("system_prompt 和 user_prompt 均不能为空")
    if not (0.0 <= temperature <= 2.0):
        raise ValueError("temperature 必须在 0.0 到 2.0 之间")
    return {
        "model": model,
        "messages": [
            {"role": "system", "content": system_prompt.strip()},
            {"role": "user", "content": user_prompt.strip()},
        ],
        "temperature": float(temperature),
        "max_tokens": max_tokens,
        "stream": stream,
    }

# 作业 3 参考答案
import httpx
from zhishu_agent.exceptions import AuthenticationError, RateLimitError
from zhishu_agent.llm_client import LLMClient
from zhishu_agent.message import ChatMessage

def safe_ask(client: LLMClient, question: str) -> str:
    try:
        return client.chat([ChatMessage(role="user", content=question)]).content
    except AuthenticationError:
        return "错误：API Key 无效，请立即停止并检查环境变量配置。"
    except RateLimitError as exc:
        wait_sec = exc.retry_after if exc.retry_after is not None else 1.0
        return f"限流告警：请求过于频繁，建议 {wait_sec} 秒后重试。"
